// CPS Rodovias Pro - Service Worker 100% Offline const CACHE_NAME = 'cps-rodovias-cache-v5'; const PRECACHE_URLS = [ './', './index.html', './public/manifest.json', './public/icon.svg', './public/dados.csv', './public/SNVBR116SPRJ.csv', './dados.csv', './SNVBR116SPRJ.csv', './Dados_Completos_Grupo_2_all.csv', 'https://cdnjs.cloudflare.com/ajax/libs/jspdf/2.5.1/jspdf.umd.min.js' ]; self.addEventListener('install', event => { event.waitUntil( caches.open(CACHE_NAME).then(async cache => { // Tenta precachear os itens individualmente para que falha de um não impeça os outros for (const url of PRECACHE_URLS) { try { await cache.add(url); } catch (err) { console.warn('[SW] Falha ao precachear URL:', url, err); } } }).then(() => self.skipWaiting()) ); }); self.addEventListener('activate', event => { event.waitUntil( caches.keys().then(keys => { return Promise.all( keys.filter(key => key !== CACHE_NAME).map(key => caches.delete(key)) ); }).then(() => self.clients.claim()) ); }); self.addEventListener('fetch', event => { // Apenas intercepta requisições GET if (event.request.method !== 'GET') return; const url = new URL(event.request.url); // Não intercepta requisições de analytics ou esquemas não http(s) if (!url.protocol.startsWith('http')) return; event.respondWith( caches.match(event.request).then(cachedResponse => { if (cachedResponse) { // Retorna do cache instantaneamente e atualiza em segundo plano se online fetch(event.request).then(networkResponse => { if (networkResponse && networkResponse.status === 200) { caches.open(CACHE_NAME).then(cache => cache.put(event.request, networkResponse)); } }).catch(() => { // Totalmente esperado quando offline }); return cachedResponse; } // Tenta rede primeiro se não estiver em cache return fetch(event.request).then(networkResponse => { if (!networkResponse || networkResponse.status !== 200 || networkResponse.type === 'opaque') { return networkResponse; } const responseToCache = networkResponse.clone(); caches.open(CACHE_NAME).then(cache => { cache.put(event.request, responseToCache); }); return networkResponse; }).catch(async () => { // Fallback offline: se for navegação HTML, entrega a página principal if (event.request.mode === 'navigate' || event.request.headers.get('accept')?.includes('text/html')) { const mainFallback = await caches.match('./index.html') || await caches.match('./'); if (mainFallback) return mainFallback; } // Se for um CSV, tenta devolver qualquer versão em cache if (url.pathname.endsWith('.csv')) { const filename = url.pathname.split('/').pop(); const csvFallback = await caches.match(`./${filename}`) || await caches.match(`./public/${filename}`); if (csvFallback) return csvFallback; } return new Response('Offline: Recurso indisponível sem conexão.', { status: 503, statusText: 'Service Unavailable (Offline)', headers: new Headers({ 'Content-Type': 'text/plain' }) }); }); }) ); });
